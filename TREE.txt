#!/usr/bin/env python3
"""リポジトリのファイル一覧を TREE.txt に出す。中身は含めない。"""
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
SKIP = {".git", "out", "__pycache__", ".venv", "node_modules"}
EXT = {".py", ".sh", ".md", ".txt", ".tsv", ".json", ".yaml", ".yml"}


def main() -> int:
    rows = []
    for p in sorted(ROOT.rglob("*")):
        if not p.is_file() or p.suffix not in EXT:
            continue
        if any(s in p.parts for s in SKIP):
            continue
        try:
            n = sum(1 for _ in p.open(encoding="utf-8", errors="ignore"))
        except Exception:
            n = -1
        rows.append(f"{p.relative_to(ROOT)}\t{n}")
    (ROOT / "TREE.txt").write_text("\n".join(rows) + "\n", encoding="utf-8")
    print(f"{len(rows)}件 → TREE.txt")
    return 0


if __name__ == "__main__":
    sys.exit(main())
